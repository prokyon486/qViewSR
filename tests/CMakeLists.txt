find_package(Qt6 REQUIRED COMPONENTS Test)
set(APP_TEST_SOURCES ${PROJECT_SOURCES})
list(REMOVE_ITEM APP_TEST_SOURCES src/main.cpp)
list(TRANSFORM APP_TEST_SOURCES PREPEND "${PROJECT_SOURCE_DIR}/")
foreach(target IN ITEMS actionmanagertests srtests navigationtests glbtests vectortests)
    add_executable(${target} tst_${target}.cpp ${APP_TEST_SOURCES} "${PROJECT_SOURCE_DIR}/resources/resources.qrc")
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    set_target_properties(${target} PROPERTIES AUTOUIC_SEARCH_PATHS "${PROJECT_SOURCE_DIR}/src")
    target_link_libraries(${target} PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Widgets Qt6::Svg
        qviewsr_i18n Qt6::Xml Qt6::Concurrent Qt6::QuickWidgets Qt6::Quick3D Qt6::Test PkgConfig::LCMS2 ZLIB::ZLIB GIF::GIF ${PLATFORM_SPECIFIC_LIBS})
    target_compile_definitions(${target} PRIVATE "VERSION=${PROJECT_VERSION}" QT_DEPRECATED_WARNINGS QT_NO_FOREACH
        "SR_SOURCE_DIR=\"${PROJECT_SOURCE_DIR}\"")
    if(QV_DISABLE_ONLINE_VERSION_CHECK)
        target_compile_definitions(${target} PRIVATE QV_DISABLE_ONLINE_VERSION_CHECK)
    endif()
    if(UNIX AND NOT APPLE AND NOT NO_X11)
        target_compile_definitions(${target} PRIVATE X11_LOADED)
    endif()
    if(WIN32 AND NOT NO_WIN32)
        target_compile_definitions(${target} PRIVATE WIN32_LOADED)
    endif()
    if(APPLE AND NOT NO_COCOA)
        target_compile_definitions(${target} PRIVATE COCOA_LOADED)
    endif()
endforeach()
add_test(NAME ActionManagerTests COMMAND actionmanagertests)
add_test(NAME SrTests COMMAND srtests)
add_test(NAME NavigationTests COMMAND navigationtests)
add_test(NAME GlbTests COMMAND glbtests)
add_test(NAME VectorTests COMMAND vectortests)
set_tests_properties(ActionManagerTests SrTests NavigationTests GlbTests VectorTests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 180)

add_executable(vectoreditdocumenttests tst_vectoreditdocumenttests.cpp ../src/vector/edit_document.cpp
    ../src/vector/raster_contours.cpp ../src/vector/memory_budget.cpp)
target_include_directories(vectoreditdocumenttests PRIVATE "${PROJECT_SOURCE_DIR}/src")
target_link_libraries(vectoreditdocumenttests PRIVATE Qt6::Core Qt6::Gui Qt6::Xml Qt6::Svg Qt6::Test)
add_test(NAME VectorEditDocumentTests COMMAND vectoreditdocumenttests)
set_tests_properties(VectorEditDocumentTests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)

add_executable(rastercontourtests tst_rastercontourtests.cpp ../src/vector/raster_contours.cpp)
target_include_directories(rastercontourtests PRIVATE "${PROJECT_SOURCE_DIR}/src")
target_link_libraries(rastercontourtests PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
add_test(NAME RasterContourTests COMMAND rastercontourtests)
set_tests_properties(RasterContourTests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
